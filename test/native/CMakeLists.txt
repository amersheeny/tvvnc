cmake_minimum_required(VERSION 3.22)
project(tvconsole_native_tests C CXX)
get_filename_component(DEPS "${CMAKE_CURRENT_LIST_DIR}/../../native" ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WITH_LIBVNCSERVER ON CACHE BOOL "" FORCE)
set(WITH_LIBVNCCLIENT ON CACHE BOOL "" FORCE)
set(LIBVNCSERVER_INSTALL OFF CACHE BOOL "" FORCE)
foreach(option WITH_EXAMPLES WITH_TESTS WITH_SDL WITH_GTK WITH_QT WITH_FFMPEG
  WITH_PNG WITH_LIBSSHTUNNEL WITH_GNUTLS WITH_OPENSSL WITH_GCRYPT WITH_SYSTEMD
  WITH_SASL WITH_XCB WITH_WEBSOCKETS WITH_TIGHTVNC_FILETRANSFER WITH_LZO)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${DEPS}/libvncserver" vnc)
add_executable(viewer_fixture viewer_fixture.c)
target_include_directories(viewer_fixture PRIVATE "${DEPS}/libvncserver/include" "${CMAKE_CURRENT_BINARY_DIR}/vnc/include")
target_link_libraries(viewer_fixture PRIVATE vncserver)
add_executable(decoder_regressions decoder_regressions.cpp)
target_compile_features(decoder_regressions PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(decoder_regressions PRIVATE /UNDEBUG)
else()
  target_compile_options(decoder_regressions PRIVATE -UNDEBUG)
endif()
target_include_directories(decoder_regressions PRIVATE "${DEPS}/libvncserver/include"
  "${CMAKE_CURRENT_BINARY_DIR}/vnc/include" "${DEPS}/libvncserver/src/common")
target_link_libraries(decoder_regressions PRIVATE vncclient)
